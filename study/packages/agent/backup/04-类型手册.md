# pi-agent-core 类型手册

## AgentState (agent.ts)

**位置**: `src/agent.ts` (构造函数中初始化)

```typescript
interface AgentState {
  systemPrompt: string;                    // 系统提示词
  model: Model<any>;                       // 当前模型
  thinkingLevel: ThinkingLevel;           // 推理级别
  tools: AgentTool<any>[];                 // 可用工具
  messages: AgentMessage[];               // 对话历史
  isStreaming: boolean;                   // 是否流式中
  streamMessage: AssistantMessage | null; // 当前流式消息
  pendingToolCalls: Set<string>;          // 执行中的 tool IDs
  error?: string;                         // 最后错误
}
```

| 字段 | 初始值 | 变更时机 |
|------|--------|----------|
| `messages` | `[]` | `appendMessage()`, `replaceMessages()`, `reset()` |
| `isStreaming` | `false` | `_runLoop()` 开始/结束时 |
| `streamMessage` | `null` | `message_start` / `message_end` |
| `pendingToolCalls` | `new Set()` | `tool_execution_start` / `end` |
| `error` | `undefined` | 出错时设置 |

## AgentMessage (types.ts)

**位置**: `src/types.ts`

```typescript
type AgentMessage = 
  | UserMessage 
  | AssistantMessage 
  | ToolResultMessage 
  | CustomAgentMessages[keyof CustomAgentMessages];
```

### UserMessage

```typescript
interface UserMessage {
  role: "user";
  content: (TextContent | ImageContent)[];
  timestamp: number;
}
```

### AssistantMessage

```typescript
interface AssistantMessage {
  role: "assistant";
  content: (TextContent | ImageContent | ToolCall)[];
  timestamp: number;
  api: string;                    // API 名称
  provider: string;               // Provider 名称
  model: string;                  // 模型 ID
  usage: TokenUsage;             // Token 使用统计
  stopReason: StopReason;        // 停止原因
  errorMessage?: string;         // 错误信息
}
```

### ToolResultMessage

```typescript
interface ToolResultMessage {
  role: "toolResult";
  content: (TextContent | ImageContent)[];
  timestamp: number;
  toolCallId: string;            // 关联的 tool call ID
  toolName: string;              // 工具名称
  result: AgentToolResult<any>;  // 执行结果
  isError: boolean;              // 是否错误
}
```

## AgentTool (types.ts)

**位置**: `src/types.ts:55`

```typescript
interface AgentTool<T> {
  name: string;                  // 唯一标识
  label: string;                 // 显示名称
  description: string;           // 功能描述（给 LLM）
  parameters: TSchema;           // JSON Schema
  execute: ExecuteFn<T>;         // 执行函数
}

// execute 函数签名
type ExecuteFn<T> = (
  toolCallId: string,           // 调用 ID
  params: Static<T>,            // 验证后的参数
  signal?: AbortSignal,         // 取消信号
  onUpdate?: OnUpdateFn        // 进度回调
) => Promise<AgentToolResult<T>>;
```

## AgentEvent (types.ts)

**位置**: `src/types.ts:200+`

```typescript
type AgentEvent =
  | AgentStartEvent
  | AgentEndEvent
  | TurnStartEvent
  | TurnEndEvent
  | MessageStartEvent
  | MessageUpdateEvent
  | MessageEndEvent
  | ToolExecutionStartEvent
  | ToolExecutionUpdateEvent
  | ToolExecutionEndEvent;
```

### 核心事件结构

```typescript
// Agent 生命周期
interface AgentStartEvent { type: "agent_start"; }
interface AgentEndEvent { type: "agent_end"; messages: AgentMessage[]; }

// Turn 生命周期
interface TurnStartEvent { type: "turn_start"; }
interface TurnEndEvent { type: "turn_end"; message: AssistantMessage; toolResults: ToolResultMessage[]; }

// 消息生命周期
interface MessageStartEvent { type: "message_start"; message: AgentMessage; }
interface MessageUpdateEvent { 
  type: "message_update"; 
  message: AssistantMessage; 
  assistantMessageEvent: AssistantMessageEvent; 
}
interface MessageEndEvent { type: "message_end"; message: AgentMessage; }

// 工具生命周期
interface ToolExecutionStartEvent { 
  type: "tool_execution_start"; 
  toolCallId: string; 
  toolName: string; 
  args: unknown; 
}
interface ToolExecutionEndEvent { 
  type: "tool_execution_end"; 
  toolCallId: string; 
  result: AgentToolResult<any>; 
}
```

## 类型转换流程

```
┌─────────────────────────────────────────────────────────────────┐
│                     类型转换关系图                               │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│   AgentMessage[]                                               │
│        │                                                        │
│        │ transformContext() [可选]                               │
│        ▼                                                        │
│   AgentMessage[]                                               │
│        │                                                        │
│        │ convertToLlm() [必需]                                   │
│        ▼                                                        │
│   Message[] (from pi-ai)                                       │
│        │                                                        │
│        │ streamSimple() / 自定义 streamFn                        │
│        ▼                                                        │
│   AssistantMessageEventStream                                  │
│        │                                                        │
│        │ for await...of                                         │
│        ▼                                                        │
│   AssistantMessage (final)                                     │
│        │                                                        │
│        │ 提取 toolCalls                                         │
│        ▼                                                        │
│   ToolResultMessage[]                                          │
│        │                                                        │
│        └─────────────────────────────────────────────────────►  │
│                              回到 AgentMessage[] 循环          │
└─────────────────────────────────────────────────────────────────┘
```
