### Overlay 类型

```typescript
interface OverlayOptions {
  width?: SizeValue;              // 宽度 (数字或 "50%")
  minWidth?: number;
  maxHeight?: SizeValue;

  anchor?: OverlayAnchor;         // 定位锚点
  offsetX?: number;
  offsetY?: number;

  row?: SizeValue;                // 绝对/百分比位置
  col?: SizeValue;

  margin?: OverlayMargin | number;
  visible?: (w: number, h: number) => boolean;
  nonCapturing?: boolean;         // 不捕获焦点
}

type SizeValue = number | `${number}%`;

type OverlayAnchor =
  | "center"
  | "top-left" | "top-right" | "bottom-left" | "bottom-right"
  | "top-center" | "bottom-center" | "left-center" | "right-center";

interface OverlayMargin {
  top?: number;
  right?: number;
  bottom?: number;
  left?: number;
}

interface OverlayHandle {
  hide(): void;
  setHidden(hidden: boolean): void;
  isHidden(): boolean;
  focus(): void;
  unfocus(): void;
  isFocused(): boolean;
}
```

### Key 类型

```typescript
// Key ID
const Key = {
  // 字母
  a: "a", b: "b", ... z: "z",

  // 数字
  Digit0: "0", Digit1: "1", ... Digit9: "9",

  // 特殊键
  Enter: "Enter",
  Escape: "Escape",
  Tab: "Tab",
  Backspace: "Backspace",
  Delete: "Delete",
  Space: "Space",

  // 方向键
  Up: "Up", Down: "Down", Left: "Left", Right: "Right",

  // Home/End
  Home: "Home", End: "End",
  PageUp: "PageUp", PageDown: "PageDown",

  // F 键
  F: (n: 1..12) => `F${n}`,

  // 修饰组合
  Ctrl: (key: string) => `Ctrl+${key}`,
  Shift: (key: string) => `Shift+${key}`,
  Alt: (key: string) => `Alt+${key}`,

  // 组合
  Ctrl: {
    Shift: (key: string) => `Ctrl+Shift+${key}`,
    Alt: (key: string) => `Ctrl+Alt+${key}`,
  },

  // 示例
  Ctrl_c: "Ctrl+c",
  Ctrl_Shift_c: "Ctrl+Shift+c",
  Alt_x: "Alt+x",
};

// 函数
declare function parseKey(data: string): KeyId | undefined;
declare function matchesKey(data: string, keyId: KeyId): boolean;
declare function isKeyRelease(data: string): boolean;
declare function isKeyRepeat(data: string): boolean;
declare function isKittyProtocolActive(): boolean;

type KeyId = string;
```

}
