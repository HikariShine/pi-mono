Pad = Math.max(0, startCol - before.width);
  const overlayPad = Math.max(0, width - overlay.width);
  const afterPad = Math.max(0, after.targetWidth - after.width);
  
  // 4. 组合结果
  return before.text + " ".repeat(beforePad) + 
         overlay.text + " ".repeat(overlayPad) +
         after.text + " ".repeat(afterPad);
}
```
