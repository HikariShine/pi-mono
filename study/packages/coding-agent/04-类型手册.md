### 2.2 工具定义

```typescript
// extensions/types.ts:750
interface ToolDefinition {
  name: string                                    // 工具名 (唯一)
  description: string                             // 工具描述 (给 LLM)
  schema: TObject                                 // JSON Schema 参数定义
  execute: (input: any, context: ToolContext) => Promise<ToolResult>
  render?: (result: ToolResult) => string[]       // 自定义渲染 (可选)
  renderOptions?: ToolRenderOptions              // 渲染选项
  promptSnippet?: string                          // 工具提示片段
  promptGuidelines?: string[]                     // 工具使用指南
}

interface ToolResult {
  content: string                                 // 文本内容
  details?: unknown                              // 详细数据 (JSON)
  isError?: boolean                              // 是否错误
}

interface ToolContext {
  abortSignal?: AbortSignal                       // 取消信号
  update?: AgentToolUpdateCallback               // 流式更新回调
}
```

## 3. 会话管理类型

### 3.1 会话条目

```typescript
// session-manager.ts
interface SessionEntryBase {
  id: string                                      // 唯一 ID
  parentId: string | null                        // 父条目 ID (树形结构)
  timestamp: number                              // 时间戳
  type: string                                   // 条目类型
}

// 消息条目
interface SessionMessageEntry extends SessionEntryBase {
  type: "message"
  role: "user" | "assistant" | "tool_result"
  content: AgentMessage
}

// 工具调用条目
interface SessionToolCallEntry extends SessionEntryBase {
  type: "tool_call"
  toolCall: ToolCall
}

// 压缩条目
interface CompactionEntry extends SessionEntryBase {
  type: "compaction"
  summary: string                                 // 摘要文本
  cutoffIndex: number                            // 切割点索引
  originalTokenCount: number                     // 原始 token 数
  compactedTokenCount: number                    // 压缩后 token 数
}

// 分支摘要条目
interface BranchSummaryEntry extends SessionEntryBase {
  type: "branch_summary"
  summary: string
  branchRootId: string
}

// 模型变更条目
interface ModelChangeEntry extends SessionEntryBase {
  type: "model_change"
  provider: string
  modelId: string
}

// 思考等级变更条目
interface ThinkingLevelChangeEntry extends SessionEntryBase {
  type: "thinking_level_change"
  level: ThinkingLevel
}
```
